# Combat v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Melee combat against NPCs — press `a` near a goblin/rat, auto-walk to it and trade blows; damage splats + HP bars; death/respawn for both NPC and player. Server-authoritative, reuses the slice-6 tick/entity system.

**Architecture:** Protocol gains `hp`/`maxHp` on entity states, an `AttackMsg`, and `HitEvent[]` on snapshots. Server adds pure helpers (`rollDamage`, `isAdjacent`) and a combat pass in `Game.step` that runs after movement: combatants with a living target either hit (when adjacent + cooldown ready) or path one step closer; death schedules NPC respawn / player respawn at SPAWN. Client reads hp from the newest frame (un-lerped), accumulates `hits` into a fading splat list, and draws HP bars + red damage numbers.

**Tech Stack:** TypeScript, Bun (`bun test`, `bun:test`), tsc, OpenTUI renderer, monorepo packages `@termenor/protocol` / server / client.

Spec: `docs/superpowers/specs/2026-06-14-combat-slice.md`. Branch: `feat/combat`.

```typescript
import type { Point } from "./pathfinding";

/** Integer damage in [0, maxHit]. */
export function rollDamage(maxHit: number, rng: () => number): number {
  return Math.floor(rng() * (maxHit + 1));
}

/** Chebyshev distance <= 1 (same tile or 8-neighbour), positions rounded to tiles. */
export function isAdjacent(a: Point, b: Point): boolean {
  const dx = Math.abs(Math.round(a.x) - Math.round(b.x));
  const dy = Math.abs(Math.round(a.y) - Math.round(b.y));
  return Math.max(dx, dy) <= 1;
}
```

```typescript
import { PLAYER_MAX_HP, ATTACK_COOLDOWN_TICKS, RESPAWN_TICKS } from "@termenor/protocol";

// 10x10 fully-walkable flat map
const open: MapData = {
  width: 10, height: 10,
  tiles: new Array(100).fill(0),
  heights: new Array(100).fill(0),
};

function maxHitRng() { return () => 0.999; } // always rolls maxHit

test("attacking an adjacent npc reduces its hp by the rolled amount on a ready cooldown", () => {
  const g = new Game(open, { x: 0, y: 0 }, maxHitRng());
  g.addPlayer("p1");          // spawns at 0,0
  g.queueMove("p1", 0, 0);    // ensure no path

  g.step(1 / 15);             // first tick: adjacent, cooldown starts ready -> hits
  const snap = g.snapshot();
  const goblin = snap.npcs.find((n) => n.id === npcId)!;
  expect(goblin.hp).toBe(5 - 1); // goblin maxHp 5, maxHit 1 -> 1 damage
  expect(snap.hits).toContainEqual({ targetId: npcId, amount: 1, tick: snap.tick });
});

test("attack cooldown gates cadence (no damage every tick)", () => {
  const g = new Game(open, { x: 0, y: 0 }, maxHitRng());
  g.addPlayer("p1");
  g.spawnNpc("goblin", 1, 0, 2);
  const npcId = g.snapshot().npcs[0].id;
  g.attack("p1", npcId);
  g.step(1 / 15); // hit #1
  g.step(1 / 15); // cooldown not ready -> no hit
  const goblin = g.snapshot().npcs.find((n) => n.id === npcId)!;
  expect(goblin.hp).toBe(4); // only one hit landed
});

test("npc retaliates against its attacker", () => {
  const g = new Game(open, { x: 0, y: 0 }, maxHitRng());
  g.addPlayer("p1");
  g.spawnNpc("goblin", 1, 0, 2);
  const npcId = g.snapshot().npcs[0].id;
  g.attack("p1", npcId);
  // run enough ticks for the goblin's own cooldown to fire at least once
  for (let i = 0; i < ATTACK_COOLDOWN_TICKS + 2; i++) g.step(1 / 15);
  const me = g.snapshot().players.find((p) => p.id === "p1")!;
  expect(me.hp).toBeLessThan(PLAYER_MAX_HP);
});

test("npc dies, leaves the snapshot, and respawns at home with full hp after RESPAWN_TICKS", () => {
  const g = new Game(open, { x: 0, y: 0 }, maxHitRng());
  g.addPlayer("p1");
  g.spawnNpc("goblin", 1, 0, 2); // maxHp 5, we deal 1/swing
  const npcId = g.snapshot().npcs[0].id;
  g.attack("p1", npcId);
  // 5 hits at 1 dmg, one hit per (ATTACK_COOLDOWN_TICKS) -> run plenty
  for (let i = 0; i < ATTACK_COOLDOWN_TICKS * 6; i++) g.step(1 / 15);
  expect(g.snapshot().npcs.find((n) => n.id === npcId)).toBeUndefined(); // dead, absent

  for (let i = 0; i < RESPAWN_TICKS + 2; i++) g.step(1 / 15);
  const respawned = g.snapshot().npcs;
  expect(respawned).toHaveLength(1);
  expect(respawned[0]).toMatchObject({ type: "goblin", x: 1, y: 0, hp: 5, maxHp: 5 });
});

test("out-of-range attacker walks toward the target before landing a hit", () => {
  const g = new Game(open, { x: 0, y: 0 }, maxHitRng());
  g.addPlayer("p1");
  g.spawnNpc("goblin", 5, 0, 1); // far away; tight leash so it won't wander to us
  const npcId = g.snapshot().npcs[0].id;
  g.attack("p1", npcId);
  // not adjacent yet -> no damage on first tick
  g.step(1 / 15);
  expect(g.snapshot().npcs.find((n) => n.id === npcId)!.hp).toBe(5);
  // give it time to close and start hitting
  for (let i = 0; i < 40; i++) g.step(1 / 15);
  expect(g.snapshot().npcs.find((n) => n.id === npcId)!.hp).toBeLessThan(5);
});

test("player respawns at SPAWN with full hp when killed", () => {
  // goblin that hits hard: use rat type but force damage via maxHitRng and many ticks.
  const g = new Game(open, { x: 0, y: 0 }, maxHitRng());
  g.addPlayer("p1");
  g.spawnNpc("goblin", 1, 0, 2);
  const npcId = g.snapshot().npcs[0].id;

    g.step(1 / 15);
  }
  const me = g.snapshot().players.find((p) => p.id === "p1")!;
  // after any death, player is back at full hp at SPAWN (0,0)
  expect(me.hp).toBe(PLAYER_MAX_HP);

test("a targeted npc stops wandering", () => {

  g.addPlayer("p1");
  g.spawnNpc("goblin", 5, 5, 4);

  const start = g.snapshot().npcs[0];
  for (let i = 0; i < 20; i++) g.step(1 / 15);
  const now = g.snapshot().npcs.find((n) => n.id === npcId)!;

  const d1 = Math.hypot(now.x - 0, now.y - 0);
```

New `step`:
```typescript
step(dt: number): void {
  this.tick++;

  // Respawn any npc whose timer elapsed.
  for (const npc of this.npcs) {
    if (npc.deadUntil >= 0 && this.tick >= npc.deadUntil) {
      npc.x = npc.home.x; npc.y = npc.home.y;
      npc.path = [];
      npc.hp = npc.maxHp;
      npc.target = null;
      npc.attackCd = 0;
      npc.deadUntil = -1;
    }
  }

  // Movement.
  for (const p of this.players.values()) advanceAlongPath(p, SPEED * dt);
  for (const npc of this.npcs) {
    if (npc.deadUntil >= 0) continue;
    advanceAlongPath(npc, NPC_SPEED * dt);
  }

  // Wander AI — only for live, untargeting npcs that are idle.
  for (const npc of this.npcs) {
    if (npc.deadUntil >= 0) continue;
    if (npc.target) continue;            // combat overrides wander
    if (npc.path.length > 0) continue;
    if (this.tick < npc.nextWanderTick) continue;
    const target = pickWanderTarget(this.map, npc.home, npc.radius, this.rng);
    if (target === null) { npc.nextWanderTick = this.tick + Math.floor(this.rng() * 15) + 5; continue; }
    const path = findPath(this.map, { x: Math.round(npc.x), y: Math.round(npc.y) }, target);
    if (path === null) { npc.nextWanderTick = this.tick + Math.floor(this.rng() * 15) + 5; continue; }
    npc.path = path;
    npc.nextWanderTick = this.tick + Math.floor(this.rng() * 45) + 15;
  }

  // Combat pass.
  for (const p of this.players.values()) {
    if (p.attackCd > 0) p.attackCd--;
    this.combatStep(p, (id) => this.npcs.find((n) => n.id === id && n.deadUntil < 0), p.maxHpHit());
  }
  for (const npc of this.npcs) {
    if (npc.deadUntil >= 0) continue;
    if (npc.attackCd > 0) npc.attackCd--;
    this.combatStep(npc, (id) => this.players.get(id) ?? null, npc.maxHit);
  }

  // Apply deaths (after all damage this tick).
  this.resolveDeaths();
}
```

Because `Player` has no `maxHit` field, do NOT call `p.maxHpHit()` — instead pass the constant `PLAYER_MAX_HIT`. Use a single private helper `combatStep` that works for either combatant type. Implement it as:

```typescript
// Shared combat actor. `findTarget` resolves the actor's target id to a live entity (or null).
private combatStep(
  actor: { x: number; y: number; facing: Facing; path: Point[]; target: string | null; attackCd: number },
  findTarget: (id: string) => { id: string; x: number; y: number; hp: number } | null | undefined,
  maxHit: number,
): void {
  if (!actor.target) return;
  const tgt = findTarget(actor.target);
  if (!tgt) { actor.target = null; return; } // target gone/dead

  if (isAdjacent(actor, tgt)) {
    actor.path = []; // stop walking once in range
    if (actor.attackCd === 0) {
      const dmg = rollDamage(maxHit, this.rng);
      tgt.hp = Math.max(0, tgt.hp - dmg);
      actor.attackCd = ATTACK_COOLDOWN_TICKS;
      this.hits.push({ targetId: tgt.id, amount: dmg, tick: this.tick });
      // Retaliation: if the victim is an npc, it targets this actor.
      const victimNpc = this.npcs.find((n) => n.id === tgt.id);
      if (victimNpc && !victimNpc.target) {
        const actorId = this.idOf(actor);
        if (actorId) victimNpc.target = actorId;
      }
    }
  } else if (actor.path.length === 0) {
    // Path toward the target's tile; stop adjacent by dropping the final step.
    const path = findPath(
      this.map,
      { x: Math.round(actor.x), y: Math.round(actor.y) },
      { x: Math.round(tgt.x), y: Math.round(tgt.y) },
    );
    if (path && path.length > 0) {
```
